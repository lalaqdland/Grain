   ```python
   from docx import Document
   
   doc = Document(file_path)
   paragraphs = []
   for para in doc.paragraphs:
       if para.text.strip():
           paragraphs.append({
               "id": generate_id(),
               "text": para.text,
               "style": para.style.name
           })
   ```
